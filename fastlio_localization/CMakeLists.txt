cmake_minimum_required(VERSION 3.0.2)
project(fastlio_localization)
find_package(catkin REQUIRED)
catkin_package()
catkin_install_python(PROGRAMS scripts/localization.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES LICENSE NOTICE DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(test/localization.test)
endif()
